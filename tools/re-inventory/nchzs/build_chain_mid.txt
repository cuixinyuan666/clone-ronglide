 2. 派发器 0x1000f680 主结构（伪码化）
================================================================================================
int buildSegment(Engine* eng /*ecx*/, int i, int n, int mode)   // ret 0xc
{
    if (C1.count >= C1.cap) return -1;                        // 0x1000f691  容量门
    rec = &C1[C1.count]; memset(rec,0,0x5d);                  // 0x1000f6ac
    rec->mode(+0x00) = mode;  rec->self(+0x04) = C1.count;    // 0x1000f6bf / f6c8
    r = seg_init(rec, eng, i, n);                             // #1
    if (r < 0) return -1;
    seg_finalize(eng, rec);                                   // #7
    for (k = r + 1; k < n; ) {
        e = seg_extend(rec, eng, k);                          // #2  (>0 == 转折点)
        if (e > 0) {
            seg_advance(eng, rec);                            // #8
            if (seg_endcheck(eng, rec, &flag) > 0)            // #3
                return seg_wrapup(eng, rec, flag);            // #4
            break;
        }
        seg_advance(eng, rec);                                // 0x1000f73a
        if (c2_reg_up(eng, rec, k)   > 0) goto L_up;          // #5
        if (c2_reg_down(eng, rec, k) > 0) goto L_down;        // #6
        if (C0[k].dir != rec->dir) break;
        k++;
    }
L_up:   if (rec->dir ==  1) return seg_wrapup(eng, rec, 1);   // 0x1000f7ed-7fd
L_down: if (rec->dir == -1) return seg_wrapup(eng, rec, 1);   // 0x1000f842-850
L_tail: rec->flags |= 1; return <endBar 相关值>;               // 0x1000f800 / f853 / f8a5  ★ 段有效标记
}
