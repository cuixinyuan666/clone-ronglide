[4] mode 1 / 2（段内扫描，[A]）
--------------------------------------------------------------------------------
 for (each C1 rec) {                                  ; ecx = rec+8, step 0x5d, trip = C1.count
   if (!(rec.flags(+0x14) & 1)) continue;
   S = rec.startBar(+0x08); E = rec.endBar(+0x0c);
   if (E-S+1 <= 3) continue;                          ; setg，等价 span>=4
   if (rec.dir(+0x10) != ∓1) continue;                ; mode1 要 -1，mode2 要 +1
   if (S+3 > E) continue;                             ; 与 span>=4 等价（冗余，编译产物）
   for (i = S+3; i <= E; i += 2) {
     if (C0[i].mark(+0x0c) & 1) continue;
     mode1: if (!(rec.ZG(+0x19) > C0[i].p2(+0x21))) continue;   ; test ah,0x41 / jne  @0x10004985
            if (!(C0[S].p2      > C0[i].p2))      continue;    ; test ah,0x05 / jp   @0x10004997
            out[C0[i].idxB(+0x15)] = 1.0;
     mode2: if (!(rec.ZD(+0x25) < C0[i].p1(+0x19))) continue;   ; test ah,0x05 / jp   @0x10004a20
            if (!(C0[S].p1      < C0[i].p1))      continue;    ; test ah,0x41 / jne  @0x10004a32
            out[C0[i].idxA(+0x11)] = 1.0;
   }
 }
