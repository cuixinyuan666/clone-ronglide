  int build(engine) {
     if (!C0.count || !C0.base || !engine[+0x28]) return -1;
     n = C0.count;
     resize(&C1, n*2); memset(C1.base, 0, n*0xba); C1.count = 0;
     resize(&C2, n);   memset(C2.base, 0, n*0x18);
     resize(&C3, n);   memset(C3.base, 0, n*0x1d);
     reset(&engine[+0x58]);
     for (i = 0; i < n; ) { r = buildSegment(engine,i,n,0);  i = (r>0 ? r : i) + 1; }   ══▶ 层 4
     /* C1 → C3 后处理: 对每个有效段 rec, 取 idx = rec->startBar+1,
        把 C0[idx].p1/p2 与 dir 映射进 C3[idx] */
     return 0;
  }
