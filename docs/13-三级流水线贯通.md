## 二、MarkSignalFlag —— sub_10060aa0

### 2.1 函数签名

```c
void MarkSignalFlag(VectorLevelObj* Wm0, float* arr, int n);
```

- `edi = Wm0`（`W->m[0]`，级别对象 vector）
- `arr` 与 `n` 在栈参数/寄存器中传递

### 2.2 指令级还原

#### 阶段 1：清零所有 `signal_flag`

```asm
0x10060ad0: mov ebx, dword ptr [edi + 8]      ; ebx = _Mylast
0x10060ad3: cmp dword ptr [edi + 4], ebx
0x10060ad6: jbe 0x10060ae0                    ; 范围断言
0x10060ae0: test edi, edi
0x10060ae2: je 0x10060ae8                     ; 空容器检查
0x10060af0: cmp esi, ebx
0x10060af2: je 0x10060b1a                     ; 遍历结束？
0x10060b01: mov dword ptr [esi + 0x8c], 0     ; ★ signal_flag = 0
0x10060b12: add esi, 0xa0                     ; 下一元素
0x10060b18: jmp 0x10060ad0                    ; 循环
```

**语义**：遍历 `W->m[0]` 的所有元素，将 `+0x8C`（`signal_flag`）清零。

#### 阶段 2：遍历 `arr[]`，按值映射

```asm
; 外层循环头（i = [esp+0x10]）
0x10060b30: fld1
0x10060b3a: fcomp dword ptr [ecx + edx*4]     ; arr[i] == 1.0f ?
0x10060b3d: fnstsw ax
0x10060b3f: test ah, 0x44
0x10060b42: jnp 0x10060b58                    ; 不等 → 到 -1.0f 检查

; arr[i] == 1.0f 分支：进入内层循环
0x10060b58: mov ebx, [edi+8]                  ; 重新加载 _Mylast
...
0x10060b89: mov ecx, [esi]                    ; ecx = obj->index
0x10060b8f: sub ecx, 1
0x10060b92: cmp eax, ecx                      ; i == index-1 ?
0x10060b94: je 0x10060ba5                     ; 找到 index == i+1
...
0x10060b9d: add esi, 0xa0                     ; 下一对象
0x10060ba3: jmp 0x10060b58                    ; 内层循环

; 找到后：
0x10060ba5: fld1
0x10060bab: fcomp dword ptr [edx + eax*4]     ; 再比较 arr[i] == 1.0f
0x10060bae: fnstsw ax
0x10060bb0: test ah, 0x44
0x10060bb3: jp 0x10060bc8                     ; 不等 → signal_flag = -1
0x10060bbc: mov dword ptr [esi + 0x8c], 1     ; ★ signal_flag = 1
0x10060bc6: jmp 0x10060bd9
0x10060bc8: ...
0x10060bcf: mov dword ptr [esi + 0x8c], 0xffffffff  ; ★ signal_flag = -1

; 外层循环尾
0x10060bd9: mov eax, [esp+0x10]
0x10060bdd: add eax, 1                        ; i++
0x10060be4: mov [esp+0x10], eax
0x10060be8: jl 0x10060b30                     ; i < n 继续
```

关键分支分析：

| `arr[i]` | `fld1 fcomp` | `fld(-1) fcomp` | 路径 |
|---|---|---|---|
| `== 1.0f` | `jnp` 跳 → `0x10060b58` | 不执行 | 到内层循环，`signal_flag = 1` |
| `== -1.0f` | `jnp` 不跳 | `jp` 不跳 → `0x10060b58` | 到内层循环，`signal_flag = -1` |
| 其他值 | `jnp` 不跳 | `jp` 跳 → `0x10060bd9` | **直接跳过**（保持 0） |

### 2.3 正确伪码

```c
void MarkSignalFlag(VectorLevelObj* Wm0, float* arr, int n)
{
    /* 1. 清零 */
    for (LevelObj* o = Wm0->_Myfirst; o != Wm0->_Mylast; ++o)
        o->signal_flag = 0;

    /* 2. 按分型序列映射 */
    for (int i = 0; i < n; i++) {
        if (arr[i] != 1.0f && arr[i] != -1.0f)
            continue;                       /* 非分型点 → 跳过 */

        /* 查找 index == i+1 的级别对象 */
        for (LevelObj* o = Wm0->_Myfirst; o != Wm0->_Mylast; ++o) {
            if (o->index == i + 1) {
                o->signal_flag = (arr[i] == 1.0f) ? 1 : -1;
                break;
            }
        }
    }
}
```

> **重要**：`wnmr.c` 中写"`arr[i] == -1.0f` 时不打标"是**错误的** —— 实测 `-1.0f` 映射为 `signal_flag = -1`。
> 另外"连续同向判断"也不存在，映射是**直接的**，不做邻近检查。

## 三、MergeByFlag —— sub_100616b0

### 3.1 函数签名

```c
void MergeByFlag(AnalyzerCtx* W);
```

### 3.2 指令级还原

#### 阶段 1：清空 `W->m[4]`（bar vector）

```asm
0x100616b9: mov eax, [edi + 0x44]             ; W->m[4]._Myfirst
0x100616be: lea esi, [edi + 0x40]             ; esi = &W->m[4]
0x100616c5: je 0x10061710                     ; _Myfirst==NULL → 跳过
0x100616ca: sub ecx, eax                      ; _Mylast - _Myfirst
0x100616d3: sar edx, 4                        ; ÷40 = 元素个数
...
0x100616fb: cmp ebx, ebp
0x100616fd: je 0x10061710                     ; 空 vector → 跳过
0x100616ff: push ebx                          ; arg1 = _Myfirst
0x10061700: mov ebx, [esi + 8]                ; ebx = _Mylast
0x10061703: mov eax, ebp                      ; eax = _Mylast
0x10061705: call 0x1006b050                   ; ★ erase(_Myfirst, _Myfirst, _Mylast)
0x1006170d: mov [esi + 8], eax                ; _Mylast = 返回值 = _Myfirst
```

`0x1006b050` 是 `vector<Bar28>::erase(first, last)` —— 将 `[first, last)` 范围内的元素删除，后面元素前移。当 `first == _Myfirst` 且 `last == _Mylast` 时，效果等价于 `clear()`。

#### 阶段 2：遍历 `W->m[0]`，从第 3 个元素开始

```asm
0x10061710: mov eax, [edi + 4]                ; W->m[0]._Myfirst
0x10061723: mov ecx, [edi + 8]                ; W->m[0]._Mylast
0x10061726: sub ecx, eax
0x1006172f: sar edx, 6                        ; ÷160 = 级别对象个数
...
0x1006173d: push 2
0x10061749: call 0x1003f0d0                   ; it = W->m[0].begin() + 2
0x10061769: jle 0x10061917                    ; n <= 2 → 直接返回
```

#### 阶段 3：主循环 — 检测相邻相反信号并合并

**路径 A**：当前 `signal_flag == 1` 且前一个 `signal_flag == -1`

```asm
0x1006178a: cmp dword ptr [edi + 0x8c], 1     ; curr->signal_flag == 1 ?
0x10061790: jne 0x1006183e                    ; 不是 → 检查路径 B
...
0x100617bd: cmp dword ptr [ebp + 0x8c], -1    ; prev->signal_flag == -1 ?
0x100617c4: jne 0x100618f5                    ; 不是 → 不合并

; 合并：取 curr->LOW 与 prev->HIGH
0x100617f8: fld dword ptr [edi + 0x0c]        ; curr->low
0x100617fe: fstp dword ptr [esp + 0x44]       ; temp_bar.low
0x10061812: fld dword ptr [ebp + 0x10]        ; prev->high
0x1006181d: fstp dword ptr [esp + 0x48]       ; temp_bar.high
0x10061821: call 0x1006a720                   ; ★ W->m[4].push_back(&temp)
```

**路径 B**：当前 `signal_flag == -1` 且前一个 `signal_flag == 1`

```asm
0x10061849: cmp dword ptr [edi + 0x8c], -1    ; curr->signal_flag == -1 ?
0x10061850: jne 0x100618f5                    ; 不是 → 不合并
...
0x1006187d: cmp dword ptr [ebp + 0x8c], 1     ; prev->signal_flag == 1 ?
0x10061884: jne 0x100618f5                    ; 不是 → 不合并

; 合并：取 prev->LOW 与 curr->HIGH
0x100618b3: fld dword ptr [ebp + 0x0c]        ; prev->low
0x100618b6: fstp dword ptr [esp + 0x44]       ; temp_bar.low
0x100618c6: fld dword ptr [edi + 0x10]        ; curr->high
0x100618d1: fstp dword ptr [esp + 0x48]       ; temp_bar.high
0x100618d5: call 0x1006a720                   ; ★ W->m[4].push_back(&temp)
```

循环尾：

```asm
0x100618ef: mov ebp, edi                      ; prev = curr
0x100618f5: add esi, 1                        ; i++
0x10061907: add edi, 0xa0                     ; curr += 0xA0
0x10061911: jl 0x10061770                     ; i < n-1 继续
```

### 3.3 正确伪码

```c
void MergeByFlag(AnalyzerCtx* W)
{
    VectorMember* m4 = &W->m[4];
    VectorMember* m0 = &W->m[0];

    /* 1. 清空 bar vector */
    if (m4->_Myfirst && m4->_Mylast > m4->_Myfirst) {
        m4->_Mylast = m4->_Myfirst;   /* 等价于 clear() */
    }

    int n = (m0->_Mylast - m0->_Myfirst) / 0xA0;
    if (n <= 2) return;

    LevelObj* prev = m0->_Myfirst + 1;   /* begin() + 1 */
    LevelObj* curr = m0->_Myfirst + 2;   /* begin() + 2 */

    for (int i = 2; i < n; i++) {
        /* 路径 A：底接顶 (1 后接 -1) */
        if (curr->signal_flag == 1 && prev->signal_flag == -1) {
            Bar28 bar;
            bar.low  = curr->low;    /* 当前低点 */
            bar.high = prev->high;   /* 前高点 */
            /* ... 其余字段按约定填充 ... */
            vec_push_back(m4, &bar);
        }
        /* 路径 B：顶接底 (-1 后接 1) */
        else if (curr->signal_flag == -1 && prev->signal_flag == 1) {
            Bar28 bar;
            bar.low  = prev->low;    /* 前低点 */
            bar.high = curr->high;   /* 当前高点 */
            /* ... */
            vec_push_back(m4, &bar);
        }

        prev = curr;
        curr += 1;   /* +0xA0 */
    }
}
```

### 3.4 关键发现

1. **`W->m[4]` 在此被填充**：之前批次的分析确认 `W->m[4]` 是 bar vector（元素尺寸 0x28），但**写入者一直未定位**。本批确认 `MergeByFlag` 是 `W->m[4]` 的主要写入者。

2. **`W->m[5]` 不参与 MergeByFlag**：`MergeByFlag` 访问的偏移只有 `+0x04`/`+0x08`（`m[0]`）和 `+0x40`/`+0x44`（`m[4]`），完全不碰 `m[5]`（`+0x50`/`+0x54`）。`m[5]` 的用途仍待后续分析。

3. **合并规则的对称性**：路径 A 与 B 完全对称，只是 LOW/HIGH 的取法互换。这保证了无论顶先还是底先，都能生成有效的 bar。

## 四、三级流水线数据流贯通

```
输入: double[] 价格序列（K线收盘价）
    ↓ 上游 CHBI（P=11/12/51）
级别对象序列 W->m[0]：含 +0x0C(LOW) +0x10(HIGH) +0x20(方向) +0x88(分型)
    ↓ WNMRNEW

┌─ MarkSignalFlag(arr[])
│    输入: 分型序列 arr[]（1.0f=顶, -1.0f=底）
│    输出: 级别对象 +0x8C(signal_flag) = {0, 1, -1}
│
├─ MergeByFlag(W)
│    输入: W->m[0]（含 signal_flag）
│    处理: 相邻相反信号 → 合并为 bar → push_back 到 W->m[4]
│    输出: W->m[4] 填充完成（Bar28 序列）
│
├─ MainEngine(W)
│    输入: W->m[4]（bar 序列）+ W->m[0]（级别对象）
│    处理: 9-bar 滑动窗口，区间不重叠判定
│    输出: 级别对象 +0x90(status=1..6) +0x94(price)
│
└─ 快照 + 输出到 VAR5/VAR6
```

### 4.1 数据流中的关键转换

| 阶段 | 输入 | 输出 | 关键操作 |
|---|---|---|---|
| CHBI | `double[]` 价格 | `LevelObj[]` | 分型、笔、线段构建 |
| MarkSignalFlag | `float[]` 分型序列 | `signal_flag ∈ {0,1,-1}` | 直接映射 |
| MergeByFlag | `LevelObj[]` + `signal_flag` | `Bar28[]` | 相邻相反 → 合并 bar |
| MainEngine | `Bar28[]` + `LevelObj[]` | `status ∈ {1..6}` | 区间不重叠判定 |

### 4.2 为何需要 MergeByFlag？

`MarkSignalFlag` 在**每个**分型点都打标（`1` 或 `-1`）。`MergeByFlag` 将相邻的顶-底对合并成一个 bar，这样 `MainEngine` 处理的是**笔级别的端点**（相邻顶底构成的区间），而非原始分型点。

这与缠论中的"笔"概念一致：顶分型与底分型之间构成一笔，笔的区间 `[low, high]` 就是合并后的 bar。
