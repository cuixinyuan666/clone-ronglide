## 二、容器模型：`std::vector<LevelObj>`（MSVC 安全 STL）

### 2.1 三处独立证据

**证据 A —— `sub_1003f0d0` 的初始化段**

```asm
0x1003f0d2: mov edi,[ebx+4]     ; edi = container->_Myfirst
0x1003f0d7: cmp edi,[ebx+8]     ; _Myfirst vs _Mylast
0x1003f0e7: mov [esi],ebx       ; iter->_Mycont = container
0x1003f0e9: mov [esi+4],edi     ; iter->_Myptr  = _Myfirst
```

**证据 B —— `sub_1006a4e0` 的拷贝构造段**

```asm
0x1006a510: mov eax,[ebx+4]     ; src->_Myfirst
0x1006a51d: mov ecx,[ebx+8]     ; src->_Mylast
0x1006a55e: mov [edi+4],eax     ; dst->_Myfirst = newbuf
0x1006a561: mov [edi+8],eax     ; dst->_Mylast  = newbuf
0x1006a564: mov [edi+0Ch],ecx   ; dst->_Myend   = newbuf + count*0xA0
```

**证据 C —— `sub_1003d720` 的边界检查**

```asm
0x1003d737: mov edx,[esi]       ; iter->_Mycont
0x1003d739: mov eax,[esi+4]     ; iter->_Myptr
0x1003d73c: cmp eax,[edx+8]     ; _Myptr vs _Mycont->_Mylast
```

### 2.2 结构体定义

```c
/* MSVC std::vector<LevelObj>（_ITERATOR_DEBUG_LEVEL >= 1，安全 STL）
 * 尺寸 16 字节 */
typedef struct VectorLevelObj {
    void*      _Myproxy;    /* +0x00  _Container_proxy*（分配器代理） */
    LevelObj*  _Myfirst;    /* +0x04  首元素指针 */
    LevelObj*  _Mylast;     /* +0x08  末元素后一位置（迭代终点） */
    LevelObj*  _Myend;      /* +0x0C  已分配容量末尾 */
} VectorLevelObj;

/* MSVC 受检迭代器 _Vector_iterator<LevelObj>（安全 STL）
 * 尺寸 8 字节 —— 普通 STL 迭代器只有 4 字节，多出的是 _Mycont */
typedef struct CheckedIter {
    VectorLevelObj* _Mycont; /* +0x00  所属容器（用于越界校验） */
    LevelObj*       _Myptr;  /* +0x04  当前元素指针 */
} CheckedIter;
```

### 2.3 安全 STL 的指纹

全函数族中反复出现 `call dword ptr ds:[1006F194h]` —— 这是 MFC80U.DLL 导入的
**`_invalid_parameter_noinfo`**。它只在 `_ITERATOR_DEBUG_LEVEL >= 1`（即
`_SECURE_SCL=1` / Debug 或启用迭代器校验的 Release）编译的 STL 中出现。
受检迭代器携带 `_Mycont` 指针正是为了在解引用时校验归属容器。

## 三、`sub_1003f0d0` 逐条还原 —— `vec.begin() + n`

### 3.1 签名与栈帧

```asm
0x1003f0d0: push ebp                  ; esp = E-4
0x1003f0d1: push edi                  ; esp = E-8
...
0x1003f113: cmp [esp+0Ch],ebp         ; [E+4] = 栈参数（唯一）
...
0x1003f144: ret 4                     ; 弹出 4 字节栈参数
```

| 项 | 值 |
|---|---|
| 容器 | `ebx`（调用方约定） |
| 输出迭代器 | `esi`（调用方约定） |
| 栈参数 `n` | `[esp+0Ch]` = `[E+4]`（入口 esp 之上） |
| 调用约定 | 混合：`ebx`/`esi` 传参 + 1 个栈参数，`ret 4` |

### 3.2 逐条语义

```asm
;; ── 阶段 1：初始化输出迭代器 = begin() ──
0x1003f0d0: push ebp
0x1003f0d1: push edi
0x1003f0d2: mov edi,[ebx+4]           ; edi = container->_Myfirst
0x1003f0d5: xor ebp,ebp               ; i = 0
0x1003f0d7: cmp edi,[ebx+8]           ; 校验 _Myfirst <= _Mylast
0x1003f0da: mov [esi],ebp             ; iter->_Mycont = 0（先置空）
0x1003f0dc: mov [esi+4],ebp           ; iter->_Myptr  = 0
0x1003f0df: jbe short 1003F0E7h
0x1003f0e1: call ds:[1006F194h]       ; 否则 _invalid_parameter_noinfo
0x1003f0e7: mov [esi],ebx             ; ★ iter->_Mycont = container
0x1003f0e9: mov [esi+4],edi           ; ★ iter->_Myptr  = begin()

;; ── 阶段 2：循环推进 n 次 ──
0x1003f0f0: mov edi,[ebx+8]           ; edi = _Mylast（迭代终点）
0x1003f0f3: cmp [ebx+4],edi           ; 再次校验
0x1003f0f6: jbe short 1003F0FEh
0x1003f0f8: call ds:[1006F194h]
0x1003f0fe: mov eax,[esi]             ; 校验 iter->_Mycont
0x1003f100: test eax,eax
0x1003f102: je short 1003F108h
0x1003f104: cmp eax,ebx
0x1003f106: je short 1003F10Eh
0x1003f108: call ds:[1006F194h]       ; 容器不匹配 → 报错
0x1003f10e: cmp [esi+4],edi           ; if (iter->_Myptr == _Mylast) 退出
0x1003f111: je short 1003F140h
0x1003f113: cmp [esp+0Ch],ebp         ; if (n == i) 退出
0x1003f117: je short 1003F140h
0x1003f119: add ebp,1                 ; i++
0x1003f11c: cmp dword ptr [esi],0     ; 校验容器非空
0x1003f11f: jne short 1003F127h
0x1003f121: call ds:[1006F194h]
0x1003f127: mov eax,[esi]
0x1003f129: mov ecx,[esi+4]
0x1003f12c: cmp ecx,[eax+8]           ; 校验 _Myptr <= _Mylast
0x1003f12f: jb short 1003F137h
0x1003f131: call ds:[1006F194h]
0x1003f137: add dword ptr [esi+4],0A0h   ; ★ iter->_Myptr += 0xA0
0x1003f13e: jmp short 1003F0F0h          ; 回到循环头

;; ── 阶段 3：返回 ──
0x1003f140: pop edi
0x1003f141: mov eax,esi               ; 返回迭代器指针
0x1003f143: pop ebp
0x1003f144: ret 4
```

### 3.3 等价 C++

```c
/* sub_1003f0d0(container /*ebx*/, out_iter /*esi*/, n /*栈*/) */
CheckedIter* vec_begin_plus_n(VectorLevelObj* container, CheckedIter* out, int n)
{
    LevelObj* p = container->_Myfirst;      /* _Myfirst */
    out->_Mycont = container;
    out->_Myptr  = p;
    for (int i = 0; ; i++) {
        if (out->_Myptr == container->_Mylast) break;   /* 到末尾 → 停止 */
        if (n == i)                            break;   /* 推进 n 次 → 停止 */
        out->_Myptr += 1;        /* +0xA0 字节，即 ++ */
    }
    return out;                 /* = container->begin() + n（越界则夹到 end） */
}
```

> **关键**：`n` **原样使用，无 ±1 调整**。`n = 0` 返回首元素，`n = k` 返回第 `k+1` 个元素（0 基）。

### 3.4 在 `sub_10056190` 中的调用模式（54 次，100% 一致）

```asm
mov  eax,[obj + 4]          ; 或 [obj + 8]
mov  ebx,[esp+43Ch]         ; ebx = 容器
push eax                    ; n = obj->field_04 / field_08
lea  esi,[esp+XXXh]         ; esi = 栈上临时迭代器
call 1003F0D0h              ; vec.begin() + n
mov  ebx,[eax]              ; iter->_Mycont
test ebx,ebx
mov  esi,[eax+4]            ; ★ esi = iter->_Myptr  → 命中的元素指针
```

| 组 | 取字段 | 语义 |
|---|---|---|
| **买入组** | `obj->+0x08` | 取「后一顶点」对象 |
| **卖出组** | `obj->+0x04` | 取「前一顶点」对象 |

## 四、`sub_1003d720` / `sub_1003d750` 逐条还原 —— 迭代器自增 / 自减

### 4.1 `sub_1003d720`（前向，`it++`）

```asm
0x1003d720: mov eax,[esi]              ; eax = iter->_Mycont
0x1003d722: test eax,eax
0x1003d724: mov ecx,[esi+4]            ; ecx = iter->_Myptr
0x1003d727: push ebx
0x1003d728: mov ebx,ds:[1006F194h]     ; _invalid_parameter_noinfo
0x1003d72e: mov [edi],eax              ; ★ out->_Mycont = 旧值（拷贝旧迭代器）
0x1003d730: mov [edi+4],ecx            ; ★ out->_Myptr  = 旧值
0x1003d733: jne short 1003D737h        ; 校验容器非空
0x1003d735: call ebx
0x1003d737: mov edx,[esi]              ; edx = _Mycont
0x1003d739: mov eax,[esi+4]            ; eax = _Myptr
0x1003d73c: cmp eax,[edx+8]            ; 校验 _Myptr <= _Mylast
0x1003d73f: jb short 1003D743h
0x1003d741: call ebx
0x1003d743: add dword ptr [esi+4],0A0h ; ★ iter->_Myptr += 0xA0
0x1003d74a: mov eax,edi                ; 返回 out（旧位置）
0x1003d74c: pop ebx
0x1003d74d: ret
```

```c
/* sub_1003d720(it /*esi*/, out /*edi*/) —— 后置自增 it++ */
CheckedIter* iter_postinc(CheckedIter* it, CheckedIter* out)
{
    *out = *it;                    /* 先拷贝旧值（后置语义） */
    assert(it->_Mycont != NULL);
    assert(it->_Myptr <= it->_Mycont->_Mylast);
    it->_Myptr += 1;               /* +0xA0 字节 */
    return out;                    /* 返回旧位置 */
}
```

### 4.2 `sub_1003d750`（后向，`it--`）

与 `sub_1003d720` **完全对称**，仅两处不同：

| 项 | `sub_1003d720`（`++`） | `sub_1003d750`（`--`） |
|---|---|---|
| 边界检查 | `cmp _Myptr, _Mylast; jb` | `cmp _Myptr, _Myfirst; ja` |
| 步进量 | `add [esi+4], 0A0h` | `add [esi+4], 0FFFFFF60h`（= **−0xA0**） |

```c
/* sub_1003d750 —— 后置自减 it-- */
CheckedIter* iter_postdec(CheckedIter* it, CheckedIter* out)
{
    *out = *it;
    assert(it->_Mycont != NULL);
    assert(it->_Myptr > it->_Mycont->_Myfirst);   /* 不能越过 begin */
    it->_Myptr -= 1;
    return out;
}
```

> `0xFFFFFF60` = −160 的二进制补码 —— **第四个独立途径印证 `sizeof(LevelObj) = 0xA0`**。

### 4.3 三函数关系图

```
            sub_1003f0d0   vec.begin() + n      （定位，n 步）
                 │
                 ▼
   ┌─────── CheckedIter{_Mycont, _Myptr} ───────┐
   │                                            │
sub_1003d720 (it++)  ────►  _Myptr += 0xA0      │  前向遍历
sub_1003d750 (it--)  ◄────  _Myptr -= 0xA0      │  后向遍历
   └────────────────────────────────────────────┘
                 │
                 ▼
        *it  →  LevelObj*（0xA0 字节对象）
```

## 五、`sizeof(LevelObj) = 0xA0` 的三途径互证

| 途径 | 位置 | 证据 |
|---|---|---|
| ① 魔数除法 | `sub_1006a4e0:0x1006a522` | `imul 66666667h; sar edx,6` → **÷160** 算元素个数 |
| ② 步进立即数 | `sub_1003d743` / `sub_1003d773` | `add [esi+4], 0A0h` / `add [esi+4], 0FFFFFF60h`（−0xA0） |
| ③ 拷贝粒度 | `sub_1006b220:0x1006b22a` | `mov ecx,28h; rep movsd` → **0x28 × 4 = 160 字节/元素** |
| ④ 分配粒度 | `sub_1003d840:0x1003d849` | `lea edx,[ecx+ecx*4]; shl edx,5` → `count × 160` |

四条路径全部指向 **160 字节**，结论稳固。

## 六、⚠️ 重要更正：`sub_1003f150` 的真实语义

### 6.1 此前（第二/第五批）的描述

```c
LevelObj* first = sub_1003f150(objList);     /* ← 标注为「取首元素」 */
for (obj = first; obj != last; obj += 0xA0)  /* 直接遍历原容器 */
```

### 6.2 实际语义：**按值拷贝容器（生成快照）**

`sub_1003f150` 全文仅 7 条指令：

```asm
0x1003f150: push ecx
0x1003f151: push esi
0x1003f152: mov dword ptr [esp+4],0    ; 置 0（保存的 ecx 槽）
0x1003f15a: call 1006A4E0h             ; ★ 转调拷贝构造函数
0x1003f15f: mov eax,esi                ; 返回隐藏返回缓冲区指针
0x1003f161: pop ecx
0x1003f162: ret
```

它**不构造迭代器**，而是转调 `sub_1006a4e0`。后者的完整行为：

```asm
0x1006a4e0: push ebp / mov ebp,esp / push -1 / push 1006C8B0h   ; SEH 序言
0x1006a50b: mov ebx,ecx                ; ebx = src（被拷贝容器）
0x1006a50d: mov edi,[ebp+8]            ; edi = 目标缓冲区（隐藏返回槽）
0x1006a510: mov eax,[ebx+4]            ; src->_Myfirst
0x1006a51d: mov ecx,[ebx+8]            ; src->_Mylast
0x1006a520: sub ecx,eax
0x1006a529: sar edx,6                  ; ÷160 → 元素个数
0x1006a542: cmp esi,1999999h
0x1006a54a: call 1003D7D0h             ; 超长 → 抛 length_error
0x1006a551: call 1003D840h             ; ★ _Allocate(count) → 堆分配 count*160
0x1006a55e: mov [edi+4],eax            ; dst->_Myfirst = 新缓冲
0x1006a561: mov [edi+8],eax            ; dst->_Mylast
0x1006a564: mov [edi+0Ch],ecx          ; dst->_Myend = 新缓冲 + count*160
0x1006a5a1: call 1006B220h             ; ★ _Uninitialized_copy 逐元素拷贝
0x1006a5a9: mov [edi+8],eax            ; dst->_Mylast = 拷贝后位置
```

**结论**：`sub_1003f150(src) → vector<LevelObj>`（按值返回一个**深拷贝**）。
即 `std::vector<LevelObj> tmp = src;`。

### 6.3 调用方证据：分配 / 释放配对

`sub_1003a7c0` 与 `sub_1003a960` 中：

```asm
;; ── 构造快照 ──
0x1003a856: lea esi,[esp+20h]        ; 隐藏返回缓冲区
0x1003a85a: mov ecx,edi              ; ecx = 源容器
0x1003a85c: call 1003F150h           ; tmp = copy(src)
0x1003a861: mov esi,[esp+24h]        ; esi = tmp->_Myfirst  ← 遍历起点
0x1003a867: cmp esi,[esp+28h]        ; 对比 tmp->_Mylast
...
;; ── 释放快照 ──
0x1003a91e: mov edx,[esp+14h]
0x1003a922: push edx
0x1003a923: call 1006B344h           ; 释放临时 float 数组
0x1003a92f: push esi
0x1003a930: call 1006B320h           ; ★ operator delete(tmp->_Myfirst) —— 释放快照
```

**`operator delete` 的存在，反证了 `sub_1003f150` 必然分配了内存** —— 若只是"取首元素"，
无需释放。这从行为上锁死了结论。

### 6.4 修正后的数据流

```c
/* WNMRNEW 的收尾（修正版） */
sub_10056190(objList);                    /* 引擎：就地标注 +0x90 / +0x94 */

VectorLevelObj snapshot = sub_1003f150(objList);   /* ★ 深拷贝一份快照 */
for (LevelObj* o = snapshot._Myfirst; o != snapshot._Mylast; ++o)
    out[o->index - 1] = (double)o->status;          /* 从快照读状态码 */

operator delete(snapshot._Myfirst);                 /* 释放快照 */
```

**功能影响**：对输出结果**无影响**（快照与原件内容相同）。差别在于
——遍历期间对原容器的修改不会影响迭代，且多了一次 O(n) 的堆分配/拷贝开销。
这解释了为何该库在长序列上存在可观测的内存抖动。

## 八、遗留问题

1. **`n` 的基数是 0 基还是 1 基** —— `sub_1003f0d0` 对 `n` 无 ±1 调整（0 基语义）。
   但 `+0x00`（序号）经 `sub esi,1` 后用作输出数组下标，**明确为 1 基**。
   若 `+0x04`/`+0x08` 同为 1 基，则 `begin()+n` 会整体偏移一位并夹到 `end`。
   两者必有一处不同 —— 需动态验证 `+0x04`/`+0x08` 的实际写入值。
2. **`+0x04`/`+0x08` 的写入点未定位** —— 已排查 `sub_1003db40`/`sub_1003ef00`，未发现；
   推测在 `sub_1003d8a0`（笔端点对象工厂）或 `sub_1003f170`（11169 条指令，未逐条展开）内。
3. **4 组的 `+0x14` 过滤条件** 未完全展开（已知 `==1` / `==-1` 两分支）。
4. **`sub_1006B320` / `sub_1006B32C`** 为 IAT 桩（`operator delete` / `operator new`），
   未在函数表中单列。
